// Computes WCAG contrast of every text-token / surface-token pair, per theme, from the CSS sources.
import { readFileSync } from "node:fs";
const SRC = process.argv[2];
const css = (f) => readFileSync(`${SRC}/${f}`, "utf8").replace(/\/\*[\s\S]*?\*\//g, "");
function block(text, selectorRe) {
  const m = text.match(selectorRe); if (!m) throw new Error("no block " + selectorRe);
  const start = text.indexOf("{", m.index) + 1; let depth = 1, i = start;
  while (depth) { if (text[i] === "{") depth++; else if (text[i] === "}") depth--; i++; }
  const vars = {}; for (const d of text.slice(start, i - 1).matchAll(/(--[\w-]+)\s*:\s*([^;]+);/g)) vars[d[1]] = d[2].trim();
  return vars;
}
function oklchToRgb(L, C, H) {
  const h = (H * Math.PI) / 180, a = C * Math.cos(h), b = C * Math.sin(h);
  const l_ = L + 0.3963377774 * a + 0.2158037573 * b, m_ = L - 0.1055613458 * a - 0.0638541728 * b, s_ = L - 0.0894841775 * a - 1.291485548 * b;
  const l = l_ ** 3, m = m_ ** 3, s = s_ ** 3;
  const lin = [4.0767416621 * l - 3.3077115913 * m + 0.2309699292 * s, -1.2684380046 * l + 2.6097574011 * m - 0.3413193965 * s, -0.0041960863 * l - 0.7034186147 * m + 1.707614701 * s];
  return lin.map((x) => Math.min(1, Math.max(0, x))).map((x) => (x <= 0.0031308 ? 12.92 * x : 1.055 * x ** (1 / 2.4) - 0.055) * 255);
}
function color(v, vars, depth = 0) {
  v = v.trim();
  const ref = v.match(/^var\((--[\w-]+)\)$/); if (ref) return vars[ref[1]] && depth < 5 ? color(vars[ref[1]], vars, depth + 1) : null;
  let m;
  if ((m = v.match(/^#([0-9a-f]{6})$/i))) return { rgb: [0, 2, 4].map((i) => parseInt(m[1].slice(i, i + 2), 16)), a: 1 };
  if ((m = v.match(/^oklch\(([\d.]+)\s+([\d.]+)\s+([\d.]+)\)$/))) return { rgb: oklchToRgb(+m[1], +m[2], +m[3]), a: 1 };
  if ((m = v.match(/^rgba?\((\d+)[ ,]+(\d+)[ ,]+(\d+)(?:\s*[,/]\s*([\d.]+))?\)$/))) return { rgb: [+m[1], +m[2], +m[3]], a: m[4] === undefined ? 1 : +m[4] };
  return null;
}
const lum = (rgb) => { const f = (x) => { x /= 255; return x <= 0.03928 ? x / 12.92 : ((x + 0.055) / 1.055) ** 2.4; }; return 0.2126 * f(rgb[0]) + 0.7152 * f(rgb[1]) + 0.0722 * f(rgb[2]); };
const over = (fg, bg) => fg.rgb.map((c, i) => c * fg.a + bg[i] * (1 - fg.a));
const ratio = (a, b) => { const x = lum(a), y = lum(b); return (Math.max(x, y) + 0.05) / (Math.min(x, y) + 0.05); };
const hex = (rgb) => "#" + rgb.map((c) => Math.round(c).toString(16).padStart(2, "0")).join("");

const base = css("styles.css"), compta = css("styles-comptabilite.css"), pme = css("styles-pme.css");
const e1L = block(base, /:root\s*\{\s*--radius:/), e1D = { ...e1L, ...block(base, /\n\.dark\s*\{/) };
const cab = { ...e1L, ...block(compta, /\n\.cabinet-workspace\s*\{\s*--background/) };
const cabD = { ...e1D, ...block(compta, /\n\.cabinet-workspace\s*\{\s*--background/) }; // no dark override exists
const pmeRoot = block(pme, /\[data-surface="pme"\]\s*\{\s*\n?\s*\/?\*?[^{]*?--bg:/);
const pmeBridge = block(pme, /\[data-surface="pme"\]\s*\{\s*--background:/);
const pmeL = { ...e1L, ...pmeRoot, ...pmeBridge };
const pmeD = { ...e1D, ...pmeRoot, ...pmeBridge, ...block(pme, /\.dark \[data-surface="pme"\]\s*\{/) };

const textE1 = ["--foreground", "--muted-foreground", "--text-2", "--primary", "--destructive", "--signal-ok", "--signal-warn", "--signal-stop", "--signal-info", "--marigold", "--card-foreground", "--accent-foreground", "--secondary-foreground"];
const surfE1 = ["--background", "--card", "--muted", "--accent", "--secondary", "--popover", "--sidebar", "--signal-soft"];
const pairsOn = [["--primary-foreground", "--primary"], ["--destructive-foreground", "--destructive"], ["--sidebar-primary-foreground", "--sidebar-primary"], ["--accent-foreground", "--accent"]];
const textPme = ["--tx", "--t2", "--t3", "--acc", "--am", "--ve", "--az", "--mg", "--me-tx", ...textE1];
const surfPme = ["--bg", "--d0", "--d1", "--d2", "--prive", "--me", "--am-d", "--az-d", "--accs", ...surfE1];
const pairsPme = [["--on-acc", "--acc"], ["--on-acc", "--acc2"], ["--me-tx", "--me"], ...pairsOn];

const out = [];
function run(theme, vars, texts, surfs, pairs) {
  const rows = [];
  const pageBg = color(vars["--bg"] || vars["--background"], vars).rgb;
  const test = (t, s) => {
    const sc = color(vars[s] || "", vars), tc = color(vars[t] || "", vars); if (!sc || !tc) return;
    const sbg = over(sc, pageBg), tfg = over(tc, sbg); const r = ratio(tfg, sbg);
    rows.push({ theme, text: t, surface: s, fg: hex(tfg), bg: hex(sbg), ratio: +r.toFixed(2), AAA: r >= 7 ? "oui" : r >= 4.5 ? "grand texte seul" : r >= 3 ? "non (AA grand seul)" : "NON" });
  };
  for (const t of texts) for (const s of surfs) test(t, s);
  for (const [t, s] of pairs) test(t, s);
  return rows;
}
out.push(...run("EditionOne clair", e1L, textE1, surfE1, pairsOn));
out.push(...run("EditionOne sombre", e1D, textE1, surfE1, pairsOn));
out.push(...run("Cabinet (clair, et aussi en .dark)", cab, textE1, surfE1, pairsOn));
out.push(...run("PME clair", pmeL, textPme, surfPme, pairsPme));
out.push(...run("PME sombre", pmeD, textPme, surfPme, pairsPme));
const fails = out.filter((r) => r.ratio < 7);
console.log(JSON.stringify({ total: out.length, sous7: fails.length, sous45: out.filter((r) => r.ratio < 4.5).length }));
console.log(JSON.stringify(out));
