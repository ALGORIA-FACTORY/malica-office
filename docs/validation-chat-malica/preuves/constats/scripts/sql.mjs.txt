// node sql.mjs <racine> <chat|local-lecture> "<sql>"  — n'imprime aucune URL.
// local-lecture : transaction READ ONLY sur la base de travail, rien n'y est écrit.
import { readFileSync } from "node:fs";
import { createRequire } from "node:module";
import { join } from "node:path";
const [RACINE, cible, sql] = process.argv.slice(2);
const require = createRequire(join(RACINE, "apps", "api", "package.json"));
const pg = require("pg");
const fichier = cible === "chat" ? ".env.test" : ".env";
const url = readFileSync(join(RACINE, "apps", "api", fichier), "utf8").split(/\r?\n/)
  .find((l) => l.startsWith("DATABASE_URL_ADMIN=")).slice(19).trim().replace(/^"|"$/g, "");
const c = new pg.Client({ connectionString: url });
await c.connect();
console.log("base:", (await c.query("select current_database() d")).rows[0].d);
if (cible !== "chat") await c.query("begin read only");
try {
  const r = await c.query(sql);
  for (const res of Array.isArray(r) ? r : [r]) console.table(res.rows);
} finally {
  if (cible !== "chat") await c.query("rollback");
  await c.end();
}
